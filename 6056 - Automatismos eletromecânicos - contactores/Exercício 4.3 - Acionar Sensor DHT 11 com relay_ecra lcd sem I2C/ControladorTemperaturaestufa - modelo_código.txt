*
  ---------------------------------------------------------
  Termóstato Digital com DHT22 e LCD 16x2 (I²C)
  ---------------------------------------------------------
  Função:
   - Mede a temperatura e a humidade com o sensor DHT22.
   - Mostra os valores no LCD e envia para o Monitor Série.
   - Controla uma saída digital (D3) com histerese:
        * Desliga acima de T1 (temperatura máxima)
        * Liga abaixo de T0 (temperatura mínima)
        * Mantém o estado dentro da banda de histerese.

  Autor: Artur G.
  Data: Outubro de 2025 versão 1.2
  ---------------------------------------------------------
*/

#include "DHT.h"
#include <LiquidCrystal_I2C.h>

// --- Configuração do LCD e do DHT ---
LiquidCrystal_I2C lcd(0x27, 16, 2);  // Endereço I²C (0x27 é o mais comum)
#define DHTPIN   6                   // Pino digital do sinal do DHT22
#define DHTTYPE  DHT22               // Tipo de sensor
DHT dht(DHTPIN, DHTTYPE);

// --- Saída digital controlada por temperatura ---
#define RELAY_PIN 3   // Saída D3 — pode comandar LED, relé, etc.

// --- Variáveis de setpoint e histerese ---
float T0 = 24.0;   // Temperatura mínima (liga quando estiver abaixo)
float T1 = 26.0;   // Temperatura máxima (desliga quando ultrapassa)

// --- Variável para estado da saída ---
bool aquecimento = false;

void setup() {
  Serial.begin(115200);
  Serial.println(F("Termóstato Digital - DHT22"));
  Serial.println(F("Autor: Artur G."));

  dht.begin();

  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW); // começa desligado

  lcd.init();
  lcd.backlight();
  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print(F("Termostato DHT22"));
  lcd.setCursor(0, 1);
  lcd.print(F("Autor: Artur G."));
  delay(1500);
  lcd.clear();
}

void loop() {
  float temperatura = dht.readTemperature();  // ºC
  float humidade    = dht.readHumidity();     // %

  if (isnan(temperatura) || isnan(humidade)) {
    Serial.println(F("Erro de leitura no DHT22"));
    lcd.setCursor(0, 0);
    lcd.print(F("Erro no sensor  "));
    lcd.setCursor(0, 1);
    lcd.print(F("Verifique cabos "));
    delay(2000);
    return;
  }

  // --- Controlo com histerese ---
  if (temperatura > T1 && aquecimento) {
    aquecimento = false;
    digitalWrite(RELAY_PIN, LOW);
  }
  else if (temperatura < T0 && !aquecimento) {
    aquecimento = true;
    digitalWrite(RELAY_PIN, HIGH);
  }
  // dentro da banda [T0, T1], mantém estado anterior

  // --- Saída série ---
  Serial.print(F("Temp: "));
  Serial.print(temperatura, 1);
  Serial.print(F(" ºC  Hum: "));
  Serial.print(humidade, 1);
  Serial.print(F("%  Saída: "));
  Serial.println(aquecimento ? F("ON") : F("OFF"));

  // --- LCD 16x2 ---
  lcd.setCursor(0, 0);
  lcd.print(F("T:"));
  lcd.print(temperatura, 1);
  lcd.print((char)223);
  lcd.print("C ");
  lcd.setCursor(9, 0);
  lcd.print(F("H:"));
  lcd.print(humidade, 0);
  lcd.print("% ");

  lcd.setCursor(0, 1);
  lcd.print(F("Saida D3: "));
  lcd.print(aquecimento ? F("ON ") : F("OFF"));

  delay(1000); // intervalo entre leituras
}

/*
  ---------------------------------------------------------
  Notas técnicas:
   - Mudar T0 e T1 conforme o intervalo desejado.
   - RELAY_PIN (D3) pode comandar um LED, transistor ou relé.
   - Exemplo: para comando de aquecimento:
       T0 = 22.0;   // Liga se < 22 ºC
       T1 = 24.0;   // Desliga se > 24 ºC
   - Assim evita oscilações rápidas e garante estabilidade.
  ---------------------------------------------------------
*/
